# Đo lõi phát (D1 bước 1) - KHÔNG nằm trong app. Dựng:  cmake -S . -B <dir> -G Ninja ... ; hai file chạy:
#   audiobench_mf  : chỉ Media Foundation (không DLL Qt Multimedia)
#   audiobench_qt  : cả hai lõi (QMediaPlayer: FFmpeg hay bộ của Windows theo QT_MEDIA_BACKEND)
cmake_minimum_required(VERSION 3.21)
project(audiobench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.10 REQUIRED COMPONENTS Gui Multimedia)
set(SRC ${CMAKE_CURRENT_LIST_DIR}/../../src)
add_executable(audiobench_mf bench_main.cpp ${SRC}/audioout.cpp ${SRC}/audioout_mf.cpp ${SRC}/audioout.h ${SRC}/audioout_mf.h)
target_compile_definitions(audiobench_mf PRIVATE NO_QT_ENGINE)
target_include_directories(audiobench_mf PRIVATE ${SRC})
target_link_libraries(audiobench_mf PRIVATE Qt6::Gui mfplat mf mfuuid ole32 ksuser propsys psapi)
add_executable(audiobench_qt bench_main.cpp audioout_qt.cpp audioout_qt.h ${SRC}/audioout.cpp ${SRC}/audioout_mf.cpp ${SRC}/audioout.h ${SRC}/audioout_mf.h)
target_compile_definitions(audiobench_qt PRIVATE)
target_include_directories(audiobench_qt PRIVATE ${SRC} ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(audiobench_qt PRIVATE Qt6::Gui Qt6::Multimedia mfplat mf mfuuid ole32 ksuser propsys psapi)
if(MSVC)
    target_compile_options(audiobench_mf PRIVATE /W3 /utf-8)
    target_compile_options(audiobench_qt PRIVATE /W3 /utf-8)
endif()
